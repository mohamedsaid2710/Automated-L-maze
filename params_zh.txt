# =============================================================================
# L-maze pseudoword generation parameters - Chinese (zh)
# =============================================================================
# Format: `key: value`, `#` starts a comment. Copy this file to make a
# variant (e.g. a stricter lexicon check) or a new language, then pass it with -p.

# Chinese (simplified). No spaces in Chinese: the input sentences must be segmented into words with
# spaces. Units are characters; candidates combine a real word-initial character with a real
# word-final one (positional model, ngram_order = 1), the usual kind of nonword in Chinese lexical
# decision. One-character words cannot get a pseudoword this way (open problem; see the plan).

# --- Language ---
language: "zh"
language_name: "Chinese"
script: "Han"
direction: "ltr"
has_case: False
lowercase: False
unit: "char"
normalise: "NFC"

# --- Lexicon (wordfreq) ---
wordlist: "large"
train_min_zipf: 2.5

# --- Language-specific generator settings ---
ngram_order: 1
min_ngram_count: 5

# --- Generator (defaults; change to tune) ---
n_samples: 400            # candidates sampled per word; more = better matches, slower
temperature: 0.5          # < 1 favours rarer letter sequences, for more varied pseudowords
min_edit_distance: 1      # the distractor must differ from its target in at least this many
                          # characters, counted as characters to change, add or delete
                          # (Levenshtein distance): 学生 -> 学长 = 1, 学生 -> 医院 = 2.
                          # Keeps the distractor from being a near-copy of the target.
len_tolerance: 1          # a word with no same-length pseudoword (short words) may get one that
                          # is this many units shorter or longer; 0 = never (x-x-x instead)
max_repeat: 1             # a distractor is used at most this often in the whole run
                          # (shared across an item's conditions counts once); 0 = no limit
repeat_exempt_max_len: 1  # distractors this short are exempt from max_repeat (function words
                          # recur in every sentence)
train_max_words: 100000   # most frequent words used to learn the letter sequences

# --- Output ---
first_word: "x-x-x"       # placeholder at the first position (standard Maze convention)
seed: 0                   # same input + same seed = same output
write_report: True        # also write <output>.report.csv (one row per distractor)
